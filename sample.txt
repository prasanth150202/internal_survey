<script>
let questions=[];
function render(qs){
  questions=qs;
  document.getElementById('questions').innerHTML=qs.map(item=>`
    <div class="question">
      <div class="qtitle">${item.id}. ${esc(item.q)}</div>
      ${item.a.map((a,i)=>`<label class="option"><input type="radio" name="q${item.id}" value="${i}" required><span>${esc(a)}</span></label>`).join('')}
    </div>`).join('');
}
function esc(s){return String(s).replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));}
google.script.run.withSuccessHandler(render).withFailureHandler(e=>document.getElementById('questions').textContent=e.message||'Could not load questions.').getQuestions();
document.getElementById('form').addEventListener('submit',e=>{
  e.preventDefault();
  const name=document.getElementById('name').value.trim();
  if(!name){alert('Please enter your name.');return;}
  const answers={};
  for(const q of questions){
    const selected=document.querySelector(`input[name="q${q.id}"]:checked`);
    if(!selected){document.querySelector(`[name="q${q.id}"]`).focus();document.getElementById('status').textContent='Please answer all 15 questions.';return;}
    answers[String(q.id)]=Number(selected.value);
  }
  const btn=document.getElementById('submit');btn.disabled=true;btn.textContent='Submitting…';
  google.script.run.withSuccessHandler(res=>{
    document.getElementById('status').textContent=res.message;
    document.getElementById('form').reset();btn.disabled=false;btn.textContent='Submit responses';
    window.scrollTo({top:0,behavior:'smooth'});
  }).withFailureHandler(err=>{
    document.getElementById('status').textContent=err.message||'Submission failed. Please try again.';
    btn.disabled=false;btn.textContent='Submit responses';
  }).submitResponse({name,team:document.getElementById('team').value,notes:document.getElementById('notes').value,answers});
});
</script>
